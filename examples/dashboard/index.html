from uniconn import Connection, allowlist_hook, audit_log_hook, metrics_collector_hook

async with Connection.from_uri("ssh://prod-web") as conn:  # ~/.ssh/config + ProxyCommand
    conn.add_hook("before_run", allowlist_hook(["ls", "pwd", "uptime"]))
    conn.add_hook("after_run", audit_log_hook("/var/log/uniconn_audit.jsonl"))
    metrics = metrics_collector_hook().attach(conn)

    async with conn.temp_dir() as tmp:
        res = await conn.run(["ls", "-la", tmp])
        if res:                          # bool(res) == res.ok
            print(res.text, metrics.to_dict())
